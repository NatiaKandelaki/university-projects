% Lie Detector — Live PPG + GSR + Temperature
% Packet format: [0xAA][PPG_H][PPG_L][GSR_H][GSR_L][TEMP_H][TEMP_L]
% Author: Natia

clear; clc;
delete(instrfindall);

s = serial('COM4', 'BaudRate', 9600, 'Timeout', 10);
fopen(s);

Fs       = 50;
BUF_SIZE = 100;
t        = (0 : BUF_SIZE-1) / Fs;

ppg_data  = zeros(1, BUF_SIZE);
gsr_data  = zeros(1, BUF_SIZE);
temp_data = zeros(1, BUF_SIZE);   % °C, one value per packet

f_low  = 0.5;
f_high = 4.0;
[b, a] = butter(4, [f_low f_high] / (Fs/2), 'bandpass');

fig = figure('Name', 'Lie Detector', 'NumberTitle', 'off');

% --- PPG raw ---
ax1   = subplot(4,1,1);
h_ppg = plot(ax1, t, ppg_data, 'b');
title(ax1, 'PPG — Raw');
xlabel(ax1, 'Time (s)'); ylabel(ax1, 'Voltage (V)');
ylim(ax1, [0 5]);

% --- PPG filtered + peaks ---
ax2      = subplot(4,1,2);
h_pfilt  = plot(ax2, t, ppg_data, 'g'); hold(ax2,'on');
h_peaks  = plot(ax2, NaN, NaN, 'rv', 'MarkerFaceColor','r');
title(ax2, 'PPG — Filtered');
xlabel(ax2, 'Time (s)'); ylabel(ax2, 'Voltage (V)');
legend(ax2, 'PPG filtered','Peaks');

% --- GSR ---
ax3   = subplot(4,1,3);
h_gsr = plot(ax3, t, gsr_data, 'm');
title(ax3, 'GSR — Skin Conductance');
xlabel(ax3, 'Time (s)'); ylabel(ax3, 'Voltage (V)');
ylim(ax3, [0 5]);

% --- Temperature ---
ax4    = subplot(4,1,4);
h_temp = plot(ax4, t, temp_data, 'r');
title(ax4, 'Skin Temperature (°C)');
xlabel(ax4, 'Time (s)'); ylabel(ax4, '°C');
ylim(ax4, [25 40]);

for i = 1 : BUF_SIZE

    % --- find sync byte ---
    sync = 0;
    while sync ~= hex2dec('AA')
        raw  = fread(s, 1, 'uint8');
        if isempty(raw), continue; end
        sync = raw;
    end

    % --- read 6 data bytes ---
    raw6 = fread(s, 6, 'uint8');
    if numel(raw6) < 6, continue; end

    ppg_raw  = bitor(bitshift(uint16(raw6(1)), 8), uint16(raw6(2)));
    gsr_raw  = bitor(bitshift(uint16(raw6(3)), 8), uint16(raw6(4)));
    temp_raw = typecast(uint16(bitor(bitshift(uint16(raw6(5)),8), ...
                                     uint16(raw6(6)))), 'int16');

    % convert to physical units
    ppg_data(i)  = double(ppg_raw)  * 5.0 / 1023.0;
    gsr_data(i)  = double(gsr_raw)  * 5.0 / 1023.0;
    temp_data(i) = double(temp_raw) * 0.0625;   % DS18B20: LSB = 0.0625°C

    % --- update PPG raw plot ---
    set(h_ppg, 'YData', ppg_data);

    % --- filter PPG and detect peaks ---
    if i >= 30
        ppg_filt   = filtfilt(b, a, ppg_data(1:i));
        ppg_smooth = movmean(ppg_filt, 5);

        partial        = zeros(1, BUF_SIZE);
        partial(1:i)   = ppg_smooth;
        set(h_pfilt, 'YData', partial);

        min_dist = round(0.4 * Fs);
        min_prom = 0.25 * (max(ppg_smooth) - min(ppg_smooth));

        try
            [pv, pl] = findpeaks(ppg_smooth, ...
                'MinPeakDistance',   min_dist, ...
                'MinPeakProminence', min_prom);
            set(h_peaks, 'XData', t(pl), 'YData', pv);

            if length(pl) >= 2
                rr  = diff(pl) / Fs;
                bpm = 60 ./ rr;
                bpm = bpm(bpm >= 45 & bpm <= 150);
                if ~isempty(bpm)
                    title(ax2, sprintf('PPG Filtered — BPM: %.1f', median(bpm)));
                end
            end
        catch
        end
    end

    % --- update GSR plot ---
    set(h_gsr,  'YData', gsr_data);

    % --- update Temperature plot ---
    set(h_temp, 'YData', temp_data);
    if temp_data(i) > 0
        title(ax4, sprintf('Skin Temperature — %.2f °C', temp_data(i)));
    end

    drawnow limitrate;
end

ppg_filt   = filtfilt(b, a, ppg_data);
ppg_smooth = movmean(ppg_filt, 5);

min_dist = round(0.4 * Fs);
min_prom = 0.25 * (max(ppg_smooth) - min(ppg_smooth));

[peak_vals, peak_locs] = findpeaks(ppg_smooth, ...
    'MinPeakDistance',   min_dist, ...
    'MinPeakProminence', min_prom);

% 60% amplitude rule
valid_locs = zeros(1, length(peak_locs));
valid_vals = zeros(1, length(peak_locs));
count = 0;
for k = 1 : length(peak_locs)
    if count == 0 || peak_vals(k) >= 0.6 * valid_vals(count)
        count = count + 1;
        valid_locs(count) = peak_locs(k);
        valid_vals(count) = peak_vals(k);
    end
end
peak_locs = valid_locs(1:count);
peak_vals = valid_vals(1:count);

set(h_pfilt, 'YData', ppg_smooth);
set(h_peaks, 'XData', t(peak_locs), 'YData', peak_vals);

if length(peak_locs) >= 2
    RR  = diff(peak_locs) / Fs;
    bpm = 60 ./ RR;
    bpm = bpm(bpm >= 45 & bpm <= 150);
    if ~isempty(bpm)
        fprintf('Detected %d heartbeats\n', count);
        fprintf('Average Heart Rate : %.1f BPM\n', median(bpm));
        fprintf('Average GSR        : %.3f V\n',   mean(gsr_data));
        fprintf('Average Temperature: %.2f C\n',   mean(temp_data(temp_data>0)));
    end
else
    disp('Not enough peaks detected. Check signal quality.');
end

fclose(s); delete(s); clear s;
